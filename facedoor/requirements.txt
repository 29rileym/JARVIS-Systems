# Jarvis FaceDoor backend dependencies
#
# IMPORTANT (Windows / Python 3.13): face_recognition depends on `dlib`, which
# normally compiles from source and needs CMake + Visual Studio Build Tools.
# To avoid that, install the prebuilt `dlib-bin` wheel and install
# face_recognition WITHOUT its deps so pip doesn't try to build dlib.
#
# Install in this order (pip can't express "use dlib-bin to satisfy dlib"):
#
#   python -m pip install dlib-bin
#   python -m pip install --no-deps face_recognition face_recognition_models
#   python -m pip install fastapi "uvicorn[standard]" numpy Pillow requests click "setuptools<81"
#
# Notes:
#  - `setuptools<81` is required: face_recognition_models imports pkg_resources,
#    which setuptools 81+ removed.
#  - Use `python -m pip ...` if `pip` isn't on your PATH.

face_recognition_models>=0.3.0
fastapi>=0.110
uvicorn[standard]>=0.27
numpy>=1.24
Pillow>=10.0
requests>=2.31
click>=8.0
